export type SizeMappingDetails = {
  pressure: number;
  runningLength: number;
  currentIndex: number;
  totalLength: number;
};
 
export type LaserPointerOptions = {
  /** @default 2 */
  size: number;
 
  streamline: number;
  simplify: number;
  simplifyPhase: "tail" | "output" | "input";
 
  keepHead: boolean;
 
  sizeMapping: (details: SizeMappingDetails) => number;

  }
 
  addPoint(point: Point) {

    if (lastPoint && lastPoint[0] === point[0] && lastPoint[1] === point[1]) {
      return;

      this.isFresh = false;
      this.stablePoints.push(point);
      return;

      this.stabilizeTail();

  }
 
  close() {
    this.stabilizeTail();
  }
 
  stabilizeTail() {

      throw new Error("Not implemented yet");

      this.stablePoints.push(...this.tailPoints);
      this.tailPoints = [];

  }

      (sizeOverride ?? this.options.size) *

        currentIndex: index,
        totalLength,

    }
